---
name: Grid
---
export const Grid = ({ columns = 2, children, gap, gapX, gapY, padding, paddingX, paddingY, style }) => {
  columns = columns >= 2 ? columns : 2;
  return (
    <div className={`grid grid-cols-1 sm:grid-cols-${columns} gap-[${gap}] gap-x-[${gapX}] gap-y-[${gapY}] p-[${padding}] px-[${paddingX}] py-[${paddingY}]`} style={style}>
      {children}
    </div>
  );
};

<Grid columns={2} gap="1rem">
  <div>item one</div>
  <div>item two</div>
  <div>item three</div>
  <div>item four</div>
 </Grid>
